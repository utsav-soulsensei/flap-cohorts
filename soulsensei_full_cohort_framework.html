Stage 2 — Free users (never paid anything)
▼
C2
Registered free — never attended, never paid 11,164 users
Enrolled in free session(s) but userHasJoined = false on all. No group or 1:1 payment.
Goal: first actual attendanceRevenue potential: low — need to build intent first
Who they are
Signed up with some curiosity but never showed up. Could be timing, wrong topic, or they forgot. Treat as cold leads — not warm ones.
Principle
Don't pitch paid. Re-earn their attention with a low-friction, high-curiosity hook.
Initiatives
Reframe around convenience
Most no-shows are timing issues, not disinterest. "Couldn't make it? Here's the next one — pick a time that works."
Try a completely different format
The original session may not have been the right hook. Offer something shorter, different topic, or a different practitioner.
Social proof nudge
"X people attended last week's session — here's what they said." A concrete outcome statement lowers the perceived risk of showing up.
📄 Show SQL query
-- C2: Registered free, never attended, never paid
SELECT DISTINCT cu."userId"
FROM "CourseUsers" cu
JOIN "CourseInstance" ci ON cu."courseInstanceId" = ci.id
JOIN "Course" c ON c.id = ci."courseId"
WHERE c.price = 0
  AND cu.status = 'ACTIVE'
  AND cu."userHasJoined" = false
  AND NOT EXISTS (
    SELECT 1 FROM "CourseUsers" cu2
    JOIN "CourseInstance" ci2 ON cu2."courseInstanceId" = ci2.id
    JOIN "Course" c2 ON c2.id = ci2."courseId"
    WHERE cu2."userId" = cu."userId"
      AND cu2."userHasJoined" = true
      AND c2.price = 0
  )
  AND NOT EXISTS (
    SELECT 1 FROM "CourseUsers" cu3
    JOIN "CourseInstance" ci3 ON cu3."courseInstanceId" = ci3.id
    JOIN "Course" c3 ON c3.id = ci3."courseId"
    WHERE cu3."userId" = cu."userId"
      AND cu3.status = 'ACTIVE' AND c3.price > 0
  )
  AND NOT EXISTS (
    SELECT 1 FROM "LeaderUserQuery" luq
    WHERE luq."userId" = cu."userId"
      AND luq.status = 'ACTIVE'
  );

-- C2b: Attended free, never paid (group or 1:1)
WITH free_attended AS (
  SELECT DISTINCT cu."userId"
  FROM "CourseUsers" cu
  JOIN "CourseInstance" ci ON cu."courseInstanceId" = ci.id
  JOIN "Course" c ON c.id = ci."courseId"
  WHERE c.price = 0
    AND cu.status = 'ACTIVE'
    AND cu."userHasJoined" = true
),
paid_group AS (
  SELECT DISTINCT cu."userId"
  FROM "CourseUsers" cu
  JOIN "CourseInstance" ci ON cu."courseInstanceId" = ci.id

  FROM "LeaderUserQuery" WHERE status = 'ACTIVE'

SELECT fa."userId"
FROM free_attended fa
WHERE fa."userId" NOT IN (SELECT "userId" FROM paid_group)
  AND fa."userId" NOT IN (SELECT "userId" FROM paid_1on1);

📄 Show SQL query
-- C2c: Attended exactly 1 free, silent 14+ days, never paid
WITH attended_once AS (
  SELECT cu."userId", MAX(ci."startDate") AS last_free_session
  FROM "CourseUsers" cu
  JOIN "CourseInstance" ci ON cu."courseInstanceId" = ci.id
  JOIN "Course" c ON c.id = ci."courseId"
  WHERE c.price = 0
    AND cu.status = 'ACTIVE'
    AND cu."userHasJoined" = true
  GROUP BY cu."userId"
  HAVING COUNT(DISTINCT cu."courseInstanceId") = 1
),
paid_group AS (
  SELECT DISTINCT cu."userId"
  FROM "CourseUsers" cu
  JOIN "CourseInstance" ci ON cu."courseInstanceId" = ci.id
  JOIN "Course" c ON c.id = ci."courseId"
  WHERE c.price > 0 AND cu.status = 'ACTIVE'
),
paid_1on1 AS (
  SELECT DISTINCT "userId"
  FROM "LeaderUserQuery" WHERE status = 'ACTIVE'
)
SELECT ao."userId"
FROM attended_once ao
WHERE ao."userId" NOT IN (SELECT "userId" FROM paid_group)
  AND ao."userId" NOT IN (SELECT "userId" FROM paid_1on1)
  AND ao.last_free_session <= NOW() - INTERVAL '14 days';

  JOIN "Course" c ON c.id = ci."courseId"

  FROM "LeaderUserQuery" WHERE status = 'ACTIVE'

FROM group_payments gp

  AND gp.last_group_session <= NOW() - INTERVAL '14 days';

📄 Show SQL query
-- C5: Paid both group + 1:1, last session within 60 days
WITH group_payments AS (
  SELECT cu."userId", MAX(ci."startDate") AS last_group
  FROM "CourseUsers" cu
  JOIN "CourseInstance" ci ON cu."courseInstanceId" = ci.id
  JOIN "Course" c ON c.id = ci."courseId"
  WHERE c.price > 0 AND cu.status = 'ACTIVE'
  GROUP BY cu."userId"
),
one_on_one AS (
  SELECT "userId", MAX("sessionDate") AS last_1on1
  FROM "LeaderUserQuery"
  WHERE status = 'ACTIVE'
  GROUP BY "userId"
),
combined AS (
  SELECT gp."userId",
         GREATEST(gp.last_group, o.last_1on1) AS last_any_session
  FROM group_payments gp
  JOIN one_on_one o ON gp."userId" = o."userId"
)
SELECT "userId"
FROM combined
WHERE last_any_session >= NOW() - INTERVAL '60 days';

📄 Show SQL query
-- C6: Last paid session (group or 1:1) 30–60 days ago
WITH group_payments AS (
  SELECT cu."userId", MAX(ci."startDate") AS last_group
  FROM "CourseUsers" cu
  JOIN "CourseInstance" ci ON cu."courseInstanceId" = ci.id
  JOIN "Course" c ON c.id = ci."courseId"
  WHERE c.price > 0 AND cu.status = 'ACTIVE'
  GROUP BY cu."userId"
),
one_on_one AS (
  SELECT "userId", MAX("sessionDate") AS last_1on1
  FROM "LeaderUserQuery"
  WHERE status = 'ACTIVE'
  GROUP BY "userId"
),
combined AS (
  SELECT COALESCE(gp."userId", o."userId") AS "userId",
         GREATEST(
           COALESCE(gp.last_group, '1970-01-01'),
           COALESCE(o.last_1on1,   '1970-01-01')
         ) AS last_any_session
  FROM group_payments gp
  FULL OUTER JOIN one_on_one o ON gp."userId" = o."userId"
)
SELECT "userId", last_any_session
FROM combined
WHERE last_any_session BETWEEN NOW() - INTERVAL '60 days'
                           AND NOW() - INTERVAL '30 days';
